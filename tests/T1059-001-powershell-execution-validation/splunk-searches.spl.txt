# ============================================================
# T1059.001 POWERSHELL EXECUTION VALIDATION
# Host: Renz
# Each numbered section is a separate Splunk search.
# ============================================================


# SEARCH 1 — AVAILABLE WINDOWS TELEMETRY

index=* host="Renz"
| stats count AS Events latest(_time) AS LastSeen BY source sourcetype
| convert ctime(LastSeen)
| sort - Events


# SEARCH 2 — SYSMON PROCESS-CREATION BASELINE

index=* host="Renz" earliest=-24h
(source="*Sysmon*" OR sourcetype="*Sysmon*")
| rex field=_raw "(?i)<EventID[^>]*>(?<ExtractedEventID>\d+)</EventID>"
| eval EventID=coalesce(EventCode,ExtractedEventID)
| search EventID=1
| rex field=_raw "(?i)<Data Name=['\"]Image['\"]>(?<Process>[^<]+)"
| rex field=_raw "(?i)<Data Name=['\"]CommandLine['\"]>(?<CommandLine>[^<]+)"
| rex field=_raw "(?i)<Data Name=['\"]ParentImage['\"]>(?<ParentProcess>[^<]+)"
| rex field=_raw "(?i)<Data Name=['\"]User['\"]>(?<User>[^<]+)"
| search Process="*powershell.exe"
| table _time host EventID User ParentProcess Process CommandLine
| sort - _time


# SEARCH 3 — POWERSHELL BASELINE MARKER

index=* host="Renz" "T1059_001_SPLUNK_TEST"
| table _time host source sourcetype EventCode _raw
| sort - _time


# SEARCH 4 — POWERSHELL OPERATIONAL ATOMIC EVENTS

index=* host="Renz" earliest=-24h
(source="*PowerShell*" OR sourcetype="*PowerShell*")
("AtomicRedTeam" OR "FromBase64String" OR "art-marker" OR "Set-Content")
| rex field=_raw "(?i)<EventID[^>]*>(?<ExtractedEventID>\d+)</EventID>"
| eval EventID=coalesce(EventCode,ExtractedEventID)
| rex field=_raw "(?s)<Data Name=['\"]ScriptBlockText['\"]>(?<ScriptBlock>.*?)</Data>"
| eval Evidence=coalesce(ScriptBlock,Message,_raw)
| table _time host EventID source sourcetype Evidence
| sort - _time


# SEARCH 5 — SYSMON ATOMIC PROCESS CREATION

index=* host="Renz" earliest=-24h
(source="*Sysmon*" OR sourcetype="*Sysmon*")
("powershell.exe" OR "reg.exe")
| rex field=_raw "(?i)<EventID[^>]*>(?<ExtractedEventID>\d+)</EventID>"
| eval EventID=coalesce(EventCode,ExtractedEventID)
| search EventID=1
| rex field=_raw "(?i)<Data Name=['\"]Image['\"]>(?<Process>[^<]+)"
| rex field=_raw "(?i)<Data Name=['\"]CommandLine['\"]>(?<CommandLine>[^<]+)"
| rex field=_raw "(?i)<Data Name=['\"]ParentImage['\"]>(?<ParentProcess>[^<]+)"
| rex field=_raw "(?i)<Data Name=['\"]User['\"]>(?<User>[^<]+)"
| search CommandLine="*AtomicRedTeam*"
    OR CommandLine="*FromBase64String*"
    OR CommandLine="*art-marker*"
    OR CommandLine="*reg.exe add*"
| table _time host EventID User ParentProcess Process CommandLine
| sort - _time


# SEARCH 6 — SYSMON REGISTRY ARTIFACTS

index=* host="Renz" earliest=-24h
(source="*Sysmon*" OR sourcetype="*Sysmon*")
"AtomicRedTeam"
| rex field=_raw "(?i)<EventID[^>]*>(?<ExtractedEventID>\d+)</EventID>"
| eval EventID=coalesce(EventCode,ExtractedEventID)
| search EventID IN (12,13,14)
| rex field=_raw "(?i)<Data Name=['\"]TargetObject['\"]>(?<TargetObject>[^<]+)"
| rex field=_raw "(?i)<Data Name=['\"]Details['\"]>(?<Details>[^<]+)"
| table _time host EventID TargetObject Details
| sort - _time


# SEARCH 7 — SYSMON FILE-CREATION EVENT ID 11

index=* host="Renz" earliest=-24h
(source="*Sysmon*" OR sourcetype="*Sysmon*")
"art-marker.txt"
| rex field=_raw "(?i)<EventID[^>]*>(?<ExtractedEventID>\d+)</EventID>"
| eval EventID=coalesce(EventCode,ExtractedEventID)
| search EventID=11
| rex field=_raw "(?i)<Data Name=['\"]TargetFilename['\"]>(?<TargetFile>[^<]+)"

| sort - _time


# SEARCH 8 — CORRELATED T1059.001 INVESTIGATION TIMELINE

index=* host="Renz" earliest=-24h
(
    (source="*Sysmon*" OR sourcetype="*Sysmon*")
    OR
    (source="*PowerShell*" OR sourcetype="*PowerShell*")
)
("AtomicRedTeam" OR "FromBase64String" OR "art-marker.txt" OR "Set-Content")
| rex field=_raw "(?i)<EventID[^>]*>(?<ExtractedEventID>\d+)</EventID>"
| eval EventID=coalesce(EventCode,ExtractedEventID)
| eval TelemetrySource=case(
    match(source,"(?i)Sysmon") OR match(sourcetype,"(?i)Sysmon"),"Sysmon",
    match(source,"(?i)PowerShell") OR match(sourcetype,"(?i)PowerShell"),"PowerShell",
    true(),"Other"
)
| rex field=_raw "(?i)<Data Name=['\"]Image['\"]>(?<Process>[^<]+)"
| rex field=_raw "(?i)<Data Name=['\"]CommandLine['\"]>(?<CommandLine>[^<]+)"
| rex field=_raw "(?i)<Data Name=['\"]TargetObject['\"]>(?<TargetObject>[^<]+)"
| rex field=_raw "(?i)<Data Name=['\"]TargetFilename['\"]>(?<TargetFile>[^<]+)"
| table _time host TelemetrySource EventID Process CommandLine TargetObject TargetFile
| sort 0 _time


# SEARCH 9 — DASHBOARD PANEL

index=* host="Renz" earliest=-24h
(
    (source="*Sysmon*" OR sourcetype="*Sysmon*")
    OR
    (source="*PowerShell*" OR sourcetype="*PowerShell*")
)
("AtomicRedTeam" OR "FromBase64String" OR "art-marker.txt" OR "Set-Content")
| rex field=_raw "(?i)<EventID[^>]*>(?<ExtractedEventID>\d+)</EventID>"
| eval EventID=coalesce(EventCode,ExtractedEventID)
| eval TelemetrySource=case(
    match(source,"(?i)Sysmon") OR match(sourcetype,"(?i)Sysmon"),"Sysmon",
    match(source,"(?i)PowerShell") OR match(sourcetype,"(?i)PowerShell"),"PowerShell",
    true(),"Other"
)
| stats count AS Events BY TelemetrySource EventID host
| sort - Events


# SEARCH 10 — SCHEDULED T1059.001 ALERT
# Schedule: every minute
# Search window: earliest=-5m latest=now
# Trigger: results greater than zero
# Throttle: 10 minutes by host

index=* host="Renz" earliest=-5m latest=now
(
    (
        (source="*PowerShell*" OR sourcetype="*PowerShell*")
        ("FromBase64String" OR "Invoke-Expression" OR "AtomicRedTeam")
    )
    OR
    (
        (source="*Sysmon*" OR sourcetype="*Sysmon*")
        ("FromBase64String" OR "AtomicRedTeam" OR "-EncodedCommand" OR " -enc ")
    )
)
| rex field=_raw "(?i)<EventID[^>]*>(?<ExtractedEventID>\d+)</EventID>"
| eval EventID=coalesce(EventCode,ExtractedEventID)
| eval TelemetrySource=case(
    match(source,"(?i)Sysmon") OR match(sourcetype,"(?i)Sysmon"),"Sysmon",
    match(source,"(?i)PowerShell") OR match(sourcetype,"(?i)PowerShell"),"PowerShell",
    true(),"Other"
)
| where
    (TelemetrySource="PowerShell" AND EventID IN (4103,4104))
    OR
    (TelemetrySource="Sysmon" AND EventID=1)
| rex field=_raw "(?i)<Data Name=['\"]Image['\"]>(?<Process>[^<]+)"
| rex field=_raw "(?i)<Data Name=['\"]CommandLine['\"]>(?<CommandLine>[^<]+)"
| rex field=_raw "(?i)<Data Name=['\"]User['\"]>(?<User>[^<]+)"
| eval DetectionReason=case(
    searchmatch("FromBase64String"),"Base64-decoded PowerShell",
    searchmatch("Invoke-Expression"),"Invoke-Expression execution",
    searchmatch("AtomicRedTeam"),"Atomic Red Team PowerShell test",
    searchmatch("-EncodedCommand") OR searchmatch(" -enc "),"Encoded PowerShell command",
    true(),"Suspicious PowerShell activity"
)
| stats count AS MatchingEvents
        values(EventID) AS EventIDs
        values(TelemetrySource) AS TelemetrySources
        values(DetectionReason) AS DetectionReasons
        values(User) AS Users
        values(Process) AS Processes
        values(CommandLine) AS CommandLines
        earliest(_time) AS FirstSeen
        latest(_time) AS LastSeen
        BY host
| convert ctime(FirstSeen) ctime(LastSeen)
| where MatchingEvents > 0
